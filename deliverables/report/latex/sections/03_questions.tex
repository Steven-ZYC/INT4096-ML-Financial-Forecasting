\section{Research Questions and Hypotheses}
\label{sec:questions}

\begin{description}[leftmargin=3em,style=sameline]
  \item[RQ1.] How do distinct models perform against naive persistence under
  identical, leakage-safe conditions?
  \item[RQ2.] How do complexity, tuning and market conditions affect
  generalisation?
  \item[RQ3.] How much do preprocessing and feature choices change forecast error?
  \item[RQ4.] What do examples and counterexamples teach learners?
\end{description}
